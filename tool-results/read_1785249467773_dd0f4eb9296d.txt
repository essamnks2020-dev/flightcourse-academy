    70→function lerp(a: number, b: number, f: number) {
    71→  return a + (b - a) * f
    72→}
    73→
    74→function interpolateFlight(elapsed: number): FlightState {
    75→  if (elapsed <= KEYFRAMES[0].t) return KEYFRAMES[0]
    76→  const last = KEYFRAMES[KEYFRAMES.length - 1]
    77→  if (elapsed >= last.t) return last
    78→  for (let i = 0; i < KEYFRAMES.length - 1; i++) {
    79→    const a = KEYFRAMES[i]
    80→    const b = KEYFRAMES[i + 1]
    81→    if (elapsed >= a.t && elapsed <= b.t) {
    82→      const f = (elapsed - a.t) / (b.t - a.t)
    83→      return {
    84→        airspeed: lerp(a.airspeed, b.airspeed, f),
    85→        altitude: lerp(a.altitude, b.altitude, f),
    86→        vsi: lerp(a.vsi, b.vsi, f),
    87→        heading: lerp(a.heading, b.heading, f),
    88→        pitch: lerp(a.pitch, b.pitch, f),
    89→        bank: lerp(a.bank, b.bank, f),
    90→      }
    91→    }
    92→  }
    93→  return last
    94→}

    96→/* ============================================================
    97→ * Needle / dial angle maps
    98→ *  Convention: 0deg = 12 o'clock (pointing up), positive = clockwise
    99→ * ========================================================== */
   100→
   101→const ASI_MAX = 160
   102→function asiAngle(kt: number) {
   103→  return -135 + (Math.min(Math.max(kt, 0), ASI_MAX) / ASI_MAX) * 270
   104→}
   105→// Altimeter 100s needle: full turn per 1000ft, continuous (no modulo) so it
   106→// never snaps backward at the 999->1000 boundary.
   107→function altAngle(ft: number) {
   108→  return (ft / 1000) * 360
   109→}
   110→// VSI: 0 at 9 o'clock (-90deg), +2000 at +50, -2000 at -230
   111→function vsiAngle(fpm: number) {
   112→  const c = Math.max(-2000, Math.min(2000, fpm))
   113→  return -90 + (c / 2000) * 140
   114→}
   115→// DG card rotates so current heading sits under the top lubber pointer.
   116→function dgCardAngle(h: number) {
   117→  return -h
   118→}
   119→// AI horizon: bank rotates the horizon (opposite sense), pitch translates it.
   120→function aiHorizonTransform(pitch: number, bank: number) {
   121→  return `rotate(${-bank}deg) translateY(${pitch * 2.2}px)`
   122→}
   123→// Turn coordinator: aircraft banks with roll.
   124→function tcAircraftAngle(bank: number) {
   125→  return bank
   126→}

   132→function polar(r: number, angleDeg: number) {
   133→  const rad = (angleDeg * Math.PI) / 180
   134→  return { x: 50 + r * Math.sin(rad), y: 50 - r * Math.cos(rad) }
   135→}
   136→
   137→function arcPath(r: number, a1: number, a2: number) {
   138→  const p1 = polar(r, a1)
   139→  const p2 = polar(r, a2)
   140→  const large = Math.abs(a2 - a1) > 180 ? 1 : 0
   141→  return `M ${p1.x.toFixed(2)} ${p1.y.toFixed(2)} A ${r} ${r} 0 ${large} 1 ${p2.x.toFixed(2)} ${p2.y.toFixed(2)}`
   142→}

   144→function Tick({ angle, major }: { angle: number; major?: boolean }) {
   145→  const r2 = major ? 41 : 43.5
   146→  return (
   147→    <line
   148→      x1="50"
   149→      y1="4"
   150→      x2="50"
   151→      y2={50 - r2}
   152→      transform={`rotate(${angle} 50 50)`}
   153→      className={major ? 'tick-major' : 'tick-minor'}
   154→    />
   155→  )
   156→}

   386→/* ============================================================
   387→ * Hotspot wrapper — keyboard accessible, filterable
   388→ * ========================================================== */
   389→
   390→function Hotspot({
   391→  hotspot,
   392→  selectedId,
   393→  activeCategory,
   394→  onSelect,
   395→  className,
   396→  children,
   397→}: {
   398→  hotspot: Hotspot
   399→  selectedId: string | null
   400→  activeCategory: Category | null
   401→  onSelect: (id: string) => void
   402→  className?: string
   403→  children: ReactNode
   404→}) {
   405→  const dim = activeCategory !== null && activeCategory !== hotspot.category
   406→  const active = selectedId === hotspot.id
   407→  return (
   408→    <div
   409→      role="button"
   410→      tabIndex={0}
   411→      aria-label={`${hotspot.name} — ${hotspot.category}`}
   412→      aria-pressed={active}
   413→      onClick={() => onSelect(hotspot.id)}
   414→      onKeyDown={(e) => {
   415→        if (e.key === 'Enter' || e.key === ' ') {
   416→          e.preventDefault()
   417→          onSelect(hotspot.id)
   418→        }
   419→      }}
   420→      className={`hotspot ${dim ? 'hotspot--dimmed' : ''} ${active ? 'hotspot--active' : ''} ${className ?? ''}`}
   421→    >
   422→      {children}
   423→    </div>
   424→  )
   425→}

   431→export default function CockpitExplorerView() {

   433→  const [running, setRunning] = useState(false)
   434→  const [selectedId, setSelectedId] = useState<string | null>(null)
   435→  const [activeCategory, setActiveCategory] = useState<Category | null>(null)
   436→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null)
   437→
   438→  const runDemo = useCallback(() => {
   439→    if (running || intervalRef.current) return
   440→    setRunning(true)
   441→    setSelectedId(null)
   442→    const start = performance.now()
   443→    intervalRef.current = setInterval(() => {
   444→      const elapsed = (performance.now() - start) / 1000
   445→      if (elapsed >= 14) {
   446→        setFlight(NEUTRAL)
   447→        setRunning(false)
   448→        if (intervalRef.current) {
   449→          clearInterval(intervalRef.current)
   450→          intervalRef.current = null
   451→        }
   452→        return
   453→      }
   454→      setFlight(interpolateFlight(elapsed))
   455→    }, 100)
   456→  }, [running])
   457→
   458→  useEffect(() => {
   459→    return () => {
   460→      if (intervalRef.current) clearInterval(intervalRef.current)
   461→    }
   462→  }, [])
   463→
   464→  const selectHotspot = useCallback((id: string) => {
   465→    setSelectedId((cur) => (cur === id ? null : id))
   466→  }, [])
   467→
   468→  const toggleCategory = useCallback((cat: Category) => {
   469→    setActiveCategory((cur) => (cur === cat ? null : cat))
   470→  }, [])
   471→
   472→  const selected = HOTSPOTS.find((h) => h.id === selectedId) ?? null
   473→
   474→  const landingOn = running && flight.airspeed < 95 && flight.altitude < 1200
   475→
   476→  const hotspotById = (id: string) => HOTSPOTS.find((h) => h.id === id)!
   477→
   478→  return (

   481→        <header className="cockpit-head">

   483→          <div className="legend" role="group" aria-label="Filter panel by category">
   484→            {CATEGORY_ORDER.map((cat) => (
   485→              <button
   486→                key={cat}
   487→                type="button"
   488→                onClick={() => toggleCategory(cat)}
   489→                aria-pressed={activeCategory === cat}
   490→                className={`legend-chip ${activeCategory === cat ? 'legend-chip--active' : ''}`}
   491→              >
   492→                <span
   493→                  className="legend-dot"
   494→                  aria-hidden="true"
   495→                  style={{ background: CATEGORY_COLOR[cat] }}
   496→                />
   497→                {cat}
   498→              </button>
   499→            ))}
   500→          </div>
   501→        </header>
   502→
   503→        {/* Carbon-fiber instrument panel */}
   504→        <section className="panel-carbon" aria-label="Six-pack flight instruments">
   505→          <div className="sixpack">
   506→            <Hotspot hotspot={hotspotById('asi')} selectedId={selectedId} activeCategory={activeCategory} onSelect={selectHotspot}>
   507→              <AirspeedIndicator value={flight.airspeed} />
   508→              <span className="gauge-caption">ASI</span>
   509→            </Hotspot>
   510→            <Hotspot hotspot={hotspotById('ai')} selectedId={selectedId} activeCategory={activeCategory} onSelect={selectHotspot}>
