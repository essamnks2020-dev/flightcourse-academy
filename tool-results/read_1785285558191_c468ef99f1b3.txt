    72→export function FlareTrainer({ className }: FlareTrainerProps) {
    73→  const [phase, setPhase] = React.useState<Phase>('start')
    74→  const phaseRef = React.useRef<Phase>('start')
    75→  const [paywallOpen, setPaywallOpen] = React.useState(false)
    76→  const [lastAttempt, setLastAttempt] = React.useState<Attempt | null>(null)
    77→  const [newBadges, setNewBadges] = React.useState<string[]>([])
    78→  const [bonusGranted, setBonusGranted] = React.useState(false)
    79→  const [guided, setGuided] = React.useState(true)
    80→  const guidedRef = React.useRef(true)
    81→  const [view, setView] = React.useState<'external' | 'cockpit'>('cockpit')
    82→  const viewRef = React.useRef<'external' | 'cockpit'>('cockpit')
    83→
    84→  const canvasRef = React.useRef<GameCanvasHandle>(null)
    85→  const cockpitRef = React.useRef<CockpitCanvasHandle>(null)
    86→  const clusterRef = React.useRef<InstrumentClusterHandle>(null)
    87→  const audio = React.useMemo(() => getAudio(), [])
    88→
    89→  // live game refs (scenarioObj is set after store selectors below)
    90→  const flareRef = React.useRef(false)
    91→  const rafRef = React.useRef(0)
    92→  const lastTime = React.useRef(0)
    93→  const lastSample = React.useRef(0)
    94→  const telemetryRef = React.useRef<ReturnType<typeof frameFromState>[]>([])
    95→  const endedAt = React.useRef<number | null>(null)
    96→  const touchdownAtRef = React.useRef<number | null>(null)
    97→  const recordedRef = React.useRef(false)
    98→  const lastCalloutRef = React.useRef(999)
    99→  const calloutsRef = React.useRef<RadarCallout[]>([])
   100→  const currentHintRef = React.useRef<string | null>(null)
   101→  const stable500Checked = React.useRef(false)
   102→  const stable500Result = React.useRef<boolean | null>(null)
   103→
   104→  // store selectors
   105→  const freePlays = useProgressStore((s) => s.freePlays)
   106→  const unlimited = useProgressStore((s) => s.unlimitedUnlocked)
   107→  const scenario = useProgressStore((s) => s.scenario)
   108→  const setScenario = useProgressStore((s) => s.setScenario)
   109→  const soundOn = useProgressStore((s) => s.soundOn)
   110→  const setSound = useProgressStore((s) => s.setSound)
   111→  const voiceCallouts = useProgressStore((s) => s.voiceCallouts)
   112→  const reducedMotion = useProgressStore((s) => s.reducedMotion)
   113→  const colorblindMode = useProgressStore((s) => s.colorblindMode)
   114→  const bestScore = useProgressStore((s) => s.bestScore)
   115→  const consumePlay = useProgressStore((s) => s.consumePlay)
   116→  const recordAttempt = useProgressStore((s) => s.recordAttempt)
   117→
   118→  // scenario object (depends on `scenario` from the store above)
   119→  const scenarioObj = React.useMemo(() => getScenario(scenario), [scenario])
   120→  const stateRef = React.useRef<FlightState>(createInitialState(scenarioObj))
   121→  const envRef = React.useRef<GameEnv>(createEnv(scenarioObj))
   122→
   123→  // keep refs in sync
   124→  const soundOnRef = React.useRef(soundOn)
   125→  const voiceCalloutsRef = React.useRef(voiceCallouts)
   126→  const reducedMotionRef = React.useRef(reducedMotion)
   127→  React.useEffect(() => {
   128→    phaseRef.current = phase
   129→  }, [phase])
   130→  React.useEffect(() => {
   131→    guidedRef.current = guided
   132→  }, [guided])
   133→  React.useEffect(() => {
   134→    viewRef.current = view
   135→  }, [view])
   136→  React.useEffect(() => {
   137→    soundOnRef.current = soundOn
   138→    audio.setMuted(!soundOn)
   139→  }, [soundOn, audio])
   140→  React.useEffect(() => {
   141→    voiceCalloutsRef.current = voiceCallouts
   142→  }, [voiceCallouts])
   143→  React.useEffect(() => {
   144→    reducedMotionRef.current = reducedMotion
   145→  }, [reducedMotion])
   146→
   147→  // --- render a frozen hero frame on start ---
   148→  React.useEffect(() => {
   149→    if (phase === 'start') {
   150→      const sc = getScenario(scenario)
   151→      const s = createInitialState(sc)
   152→      const e = createEnv(sc)
   153→      stateRef.current = s
   154→      envRef.current = e
   155→      const id = requestAnimationFrame(() => {
   156→        canvasRef.current?.renderFrame(s, e)
   157→        cockpitRef.current?.renderFrame(s, e)
   158→      })
   159→      return () => cancelAnimationFrame(id)
   160→    }
   161→  }, [phase, scenario])
   162→
   163→  // --- main game loop ---
   164→  const startLoop = React.useCallback(() => {
   165→    lastTime.current = performance.now()
   166→    lastSample.current = 0
   167→    telemetryRef.current = []
   168→    endedAt.current = null
   169→    recordedRef.current = false
   170→    lastCalloutRef.current = 999
   171→    calloutsRef.current = []
   172→    currentHintRef.current = null
   173→    stable500Checked.current = false
   174→    stable500Result.current = null
   175→    touchdownAtRef.current = null
   176→    const loop = (now: number) => {
   177→      if (phaseRef.current !== 'playing') return
   178→      const dt = Math.min(0.05, (now - lastTime.current) / 1000)
   179→      lastTime.current = now
   180→
   181→      const s = stepFlight(stateRef.current, flareRef.current, dt, envRef.current)
   182→      s.lastCalloutAlt = lastCalloutRef.current
   183→      stateRef.current = s
   184→      // render to whichever view is active (both kept in sync for instant toggle)
   185→      canvasRef.current?.renderFrame(s, envRef.current)
   186→      cockpitRef.current?.renderFrame(s, envRef.current)
   187→      clusterRef.current?.setAll({
   188→        altitude: s.altitude,
   189→        airspeed: s.airspeed,
   190→        vsi: s.vsi,
   191→      })
   192→
   193→      // --- audio: continuous engine + wind ---
   194→      audio.updateEngine(s.airspeed)
   195→      audio.setStallHorn(s.stallHorn && s.altitude < 30 && !s.onGround)
   196→
   197→      // --- coaching: radar callouts (guided only) ---
   198→      if (guidedRef.current) {
   199→        const callout = checkRadarCallout(s)
   200→        if (callout) {
   201→          lastCalloutRef.current = callout.alt
   202→          calloutsRef.current.push(callout)
   203→          if (!soundOnRef.current) {
   204→            // still beep if sound is on; voice handled below
   205→          }
   206→          audio.calloutBeep()
   207→          // Voice callouts via speechSynthesis (§2.4) — zero dependency
   208→          if (voiceCalloutsRef.current && typeof window !== 'undefined' && window.speechSynthesis) {
   209→            window.speechSynthesis.cancel() // never queue — always speak the latest
   210→            const u = new SpeechSynthesisUtterance(callout.text.replace(' — FLARE', ''))
   211→            u.rate = 1.3
   212→            u.volume = 0.7
   213→            u.pitch = 1.0
   214→            window.speechSynthesis.speak(u)
   215→          }
   216→        }
   217→        const hint = liveHint(s)
   218→        currentHintRef.current = hint?.text ?? null
   219→      }
   220→
   221→      // --- stable-approach check at 500ft ---
   222→      if (!stable500Checked.current && s.altitude <= 500 && s.altitude > 480) {
   223→        stable500Checked.current = true
   224→        const onSpeed = Math.abs(s.airspeed - APPROACH_SPEED) < 10
   225→        const sinkOk = Math.abs(s.vsi) < 1000
   226→        stable500Result.current = onSpeed && sinkOk
   227→      }
   228→
   229→      // sample telemetry ~20Hz
   230→      if (now - lastSample.current > 50) {
   231→        telemetryRef.current.push(frameFromState(s))
   232→        lastSample.current = now
   233→      }
   234→
   235→      // touchdown effects — fire when wheels first touch (s.onGround && s.result),
   236→      // NOT when s.ended (which now includes the rollout phase)
   237→      if (s.onGround && s.result && touchdownAtRef.current === null) {
   238→        touchdownAtRef.current = now
   239→        const sev = Math.abs(s.result.touchdownVSI)
   240→        const q = s.result.quality
   241→        if (!reducedMotionRef.current) {
   242→          if (q === 'greaser' || q === 'good' || q === 'firm') {
   243→            canvasRef.current?.burst('smoke', Math.min(1.4, sev / 240 + 0.4))
   244→          } else {
   245→            canvasRef.current?.burst('dust', Math.min(1.8, sev / 220 + 0.6))
   246→          }
   247→          canvasRef.current?.shake(Math.min(20, sev / 34))
   248→        }
   249→        audio.touchdown(Math.min(1, sev / 500))
   250→        audio.setStallHorn(false)
   251→        // Greaser chime (§1.5) — a distinct positive cue for the best outcome
   252→        if (q === 'greaser') {
   253→          audio.sfx('chime')
   254→        }
   255→      }
   256→
   257→      // The flight ends after rollout completes (s.ended). Give a beat for
   258→      // particles + the rollout to settle, then finalize.
   259→      if (s.ended && endedAt.current === null) {
   260→        endedAt.current = now
   261→      }
   262→      if (s.ended && now - (endedAt.current ?? now) > 600 && !recordedRef.current) {
   263→        recordedRef.current = true
   264→        finalizeRef.current?.()
   265→        return
   266→      }
   267→
   268→      rafRef.current = requestAnimationFrame(loop)
   269→    }
   270→    rafRef.current = requestAnimationFrame(loop)
   271→  }, [])
   272→
   273→  const finalizeAttempt = React.useCallback(() => {
   274→    const s = stateRef.current
   275→    const result = s.result
   276→    if (!result) {
   277→      setPhase('start')
   278→      return
   279→    }
   280→    const attempt: Attempt = {
   281→      id: uid(),
   282→      timestamp: Date.now(),
   283→      score: result.score,
   284→      quality: result.quality,
   285→      touchdownVSI: result.touchdownVSI,
   286→      touchdownAirspeed: result.touchdownAirspeed,
   287→      touchdownDistance: result.touchdownDistance,
   288→      touchdownLateral: result.touchdownLateral,
   289→      touchdownCrab: result.touchdownCrab,
   290→      touchdownPitch: result.touchdownPitch,
   291→      flareAltitude: result.flareAltitude,
   292→      flareTiming: result.flareTiming,
   293→      bounces: result.bounces,
   294→      stalled: result.stalled,
   295→      crosswind: result.crosswind,
   296→      duration: result.duration,
   297→      stableAt500: stable500Result.current,
   298→      maxBalloon: result.maxBalloon,
   299→      scenarioId: scenario,
   300→      telemetry: telemetryRef.current,
   301→    }
   302→    const res = recordAttempt(attempt)
   303→    setLastAttempt(attempt)
   304→    setNewBadges(res.newBadges)
   305→    setBonusGranted(res.bonusGranted)
   306→    if (res.newBadges.length > 0) {
   307→      toast.success(`Achievement unlocked: ${res.newBadges.length} new!`)
   308→    }
   309→    setPhase('result')
   310→  }, [recordAttempt])
   311→
   312→  const finalizeRef = React.useRef(finalizeAttempt)
   313→  React.useEffect(() => {
   314→    // eslint-disable-next-line react-hooks/immutability
   315→    finalizeRef.current = finalizeAttempt
   316→  }, [finalizeAttempt])
   317→
   318→  // --- controls ---
   319→  const beginFlight = React.useCallback(() => {
   320→    const ok = consumePlay()
   321→    if (!ok) {
   322→      setPaywallOpen(true)
   323→      return
   324→    }
   325→    // init audio on this user gesture (autoplay policy)
   326→    audio.init()
   327→    audio.setMuted(!soundOn)
   328→    audio.startEngine()
   329→    audio.sfx('click')
   330→    const sc = getScenario(scenario)
   331→    const s = createInitialState(sc)
   332→    const e = createEnv(sc)
   333→    stateRef.current = s
   334→    envRef.current = e
   335→    flareRef.current = false
   336→    canvasRef.current?.reset()
   337→    cockpitRef.current?.reset()
   338→    setPhase('playing')
   339→    requestAnimationFrame(() => startLoop())
   340→  }, [consumePlay, scenario, startLoop, audio])
   341→
   342→  // §2.1: "Go around" = the taught procedure — power up, climb away, good
   343→  // decision coaching toast. No landing recorded.
   344→  const goAround = React.useCallback(() => {
   345→    phaseRef.current = 'start'
   346→    cancelAnimationFrame(rafRef.current)
   347→    audio.setStallHorn(false)
   348→    audio.sfx('chime') // power-up cue
   349→    toast.info('Go around — good decision. The landing is never mandatory.')
   350→    setPhase('start')
   351→  }, [audio])
   352→
   353→  // §2.1: "Abandon" = instant restart with no scoring, no fanfare.
   354→  const abandon = React.useCallback(() => {
   355→    phaseRef.current = 'start'
   356→    cancelAnimationFrame(rafRef.current)
   357→    audio.setStallHorn(false)
   358→    setPhase('start')
   359→  }, [])
   360→
   361→  const flyAgain = React.useCallback(() => {
   362→    const ok = consumePlay()
   363→    if (!ok) {
   364→      setPaywallOpen(true)
   365→      return
   366→    }
   367→    audio.init()
   368→    audio.setMuted(!soundOn)
   369→    audio.startEngine()
   370→    audio.sfx('click')
   371→    const sc = getScenario(scenario)
   372→    const s = createInitialState(sc)
   373→    const e = createEnv(sc)
   374→    stateRef.current = s
   375→    envRef.current = e
   376→    flareRef.current = false
   377→    canvasRef.current?.reset()
   378→    cockpitRef.current?.reset()
   379→    setNewBadges([])
   380→    setBonusGranted(false)
   381→    setPhase('playing')
   382→    requestAnimationFrame(() => startLoop())
   383→  }, [consumePlay, scenario, startLoop, audio])
   384→
   385→  const shareToEarnFromPaywall = React.useCallback(() => {
   386→    if (lastAttempt) setPhase('result')
   387→    else toast.info('Complete a landing, then share it to earn a bonus play.')
   388→  }, [lastAttempt])
   389→
   390→  // --- input handlers ---
   391→  React.useEffect(() => {
   392→    const down = (e: KeyboardEvent) => {
   393→      if (e.code === 'Space' || e.key === ' ' || e.keyCode === 32) {
   394→        e.preventDefault()
   395→        if (phaseRef.current === 'playing') flareRef.current = true
   396→        else if (phaseRef.current === 'start') beginFlight()
   397→      }
   398→      if ((e.key === 'g' || e.key === 'G') && phaseRef.current === 'playing') {
   399→        goAround()
   400→      }
   401→      if (e.key === 'Escape' && phaseRef.current === 'playing') {
   402→        abandon()
   403→      }
   404→    }
   405→    const up = (e: KeyboardEvent) => {
   406→      if (e.code === 'Space' || e.key === ' ' || e.keyCode === 32) {
   407→        e.preventDefault()
   408→        flareRef.current = false
   409→      }
   410→    }
   411→    window.addEventListener('keydown', down)
   412→    window.addEventListener('keyup', up)
   413→    return () => {
   414→      window.removeEventListener('keydown', down)
   415→      window.removeEventListener('keyup', up)
   416→    }
   417→  }, [beginFlight, goAround, abandon])
   418→
   419→  React.useEffect(() => () => cancelAnimationFrame(rafRef.current), [])
   420→
   421→  const onPointerDown = () => {
   422→    if (phaseRef.current === 'playing') flareRef.current = true
   423→  }
   424→  const onPointerUp = () => {
   425→    flareRef.current = false
   426→  }
   427→
   428→  return (
   429→    <div className={cn('flex min-h-screen flex-col bg-background', className)}>
   430→      {/* top brand bar */}
   431→      <header className="z-20 flex items-center justify-between border-b border-white/10 bg-navy/70 px-4 py-2 backdrop-blur-md shadow-[0_4px_20px_rgba(0,0,0,0.3)]">
   432→        <div className="flex items-center gap-2">
   433→          <div className="flex h-7 w-7 items-center justify-center rounded-md bg-horizon-gold/15 ring-1 ring-horizon-gold/30">
   434→            <Plane className="h-4 w-4 text-horizon-gold" />
   435→          </div>
   436→          <span className="font-sora text-sm font-bold tracking-tight text-foreground sm:text-base">
   437→            Flight<span className="text-horizon-gold">Course</span>
   438→          </span>
   439→          <span className="hidden font-jetbrains text-[10px] uppercase tracking-widest text-muted-foreground sm:inline">
   440→            · Landing Flare Trainer
   441→          </span>
   442→        </div>
   443→        <div className="flex items-center gap-2">
   444→          {unlimited ? (
   445→            <UiBadge className="border-horizon-gold/40 bg-horizon-gold/15 text-horizon-gold">
   446→              <Zap className="mr-1 h-3 w-3" /> Unlimited
   447→            </UiBadge>
   448→          ) : (
   449→            <span className="font-jetbrains text-xs text-muted-foreground">
   450→              <span className={cn('font-bold', freePlays > 0 ? 'text-sky' : 'text-destructive')}>
   451→                {freePlays}
   452→              </span>{' '}
   453→              free plays
   454→            </span>
   455→          )}
   456→          {/* mute toggle */}
   457→          <Button
   458→            variant="ghost"
   459→            size="icon"
   460→            className="h-8 w-8 text-muted-foreground hover:text-foreground"
   461→            onClick={() => setSound(!soundOn)}
   462→            aria-label={soundOn ? 'Mute' : 'Unmute'}
   463→            title={soundOn ? 'Mute audio' : 'Unmute audio'}
   464→          >
   465→            {!soundOn ? <VolumeX className="h-4 w-4" /> : <Volume2 className="h-4 w-4" />}
   466→          </Button>
   467→        </div>
   468→      </header>
   469→
   470→      {/* main stage — min-height ensures content never sits under the footer */}
   471→      <main className="relative flex min-h-[calc(100vh-7.5rem)] flex-1 flex-col">
   472→        {/* both renderers always mounted; toggle visibility for instant switching */}
   473→        <div className={cn('absolute inset-0', view === 'external' ? 'z-0' : 'z-0 opacity-0 pointer-events-none')}>
   474→          <GameCanvas ref={canvasRef} className="h-full w-full" />
   475→        </div>
   476→        <div className={cn('absolute inset-0', view === 'cockpit' ? 'z-0' : 'z-0 opacity-0 pointer-events-none')}>
   477→          <CockpitCanvas ref={cockpitRef} className="h-full w-full" />
   478→        </div>
   479→
   480→        {phase === 'start' && (
   481→          <StartScreen
   482→            onBegin={beginFlight}
   483→            scenario={scenario}
   484→            onScenario={setScenario}
   485→            freePlays={freePlays}
   486→            unlimited={unlimited}
   487→            guided={guided}
   488→            onGuided={setGuided}
   489→            view={view}
   490→            onView={setView}
   491→            onOpenPaywall={() => setPaywallOpen(true)}
   492→          />
   493→        )}
   494→
   495→        {phase === 'playing' && (
   496→          <PlayingHud
   497→            clusterRef={clusterRef}
   498→            freePlays={freePlays}
   499→            unlimited={unlimited}
   500→            crosswind={envRef.current.crosswind > 0}
   501→            guided={guided}
   502→            view={view}
   503→            onView={setView}
   504→            stateRef={stateRef}
   505→            calloutsRef={calloutsRef}
   506→            hintRef={currentHintRef}
   507→            onGoAround={goAround}
   508→            onAbandon={abandon}
   509→            runwayHeading={envRef.current.runwayHeading}
   510→            colorblindMode={colorblindMode}
   511→            onPointerDown={onPointerDown}
   512→            onPointerUp={onPointerUp}
   513→          />
   514→        )}
   515→
   516→        {phase === 'result' && lastAttempt && (
   517→          <ResultScreen
   518→            attempt={lastAttempt}
   519→            bestScore={bestScore}
   520→            newBadges={newBadges}
   521→            bonusGranted={bonusGranted}
