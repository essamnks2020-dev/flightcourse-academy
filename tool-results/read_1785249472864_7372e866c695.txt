    70→    70→function lerp(a: number, b: number, f: number) {
    71→    71→  return a + (b - a) * f
    72→    72→}
    73→    73→
    74→    74→function interpolateFlight(elapsed: number): FlightState {
    75→    75→  if (elapsed <= KEYFRAMES[0].t) return KEYFRAMES[0]
    76→    76→  const last = KEYFRAMES[KEYFRAMES.length - 1]
    77→    77→  if (elapsed >= last.t) return last
    78→    78→  for (let i = 0; i < KEYFRAMES.length - 1; i++) {
    79→    79→    const a = KEYFRAMES[i]
    80→    80→    const b = KEYFRAMES[i + 1]
    81→    81→    if (elapsed >= a.t && elapsed <= b.t) {
    82→    82→      const f = (elapsed - a.t) / (b.t - a.t)
    83→    83→      return {
    84→    84→        airspeed: lerp(a.airspeed, b.airspeed, f),
    85→    85→        altitude: lerp(a.altitude, b.altitude, f),
    86→    86→        vsi: lerp(a.vsi, b.vsi, f),
    87→    87→        heading: lerp(a.heading, b.heading, f),
    88→    88→        pitch: lerp(a.pitch, b.pitch, f),
    89→    89→        bank: lerp(a.bank, b.bank, f),
    90→    90→      }
    91→    91→    }
    92→    92→  }
    93→    93→  return last
    94→    94→}

    96→    96→/* ============================================================
    97→    97→ * Needle / dial angle maps
    98→    98→ *  Convention: 0deg = 12 o'clock (pointing up), positive = clockwise
    99→    99→ * ========================================================== */
   100→   100→
   101→   101→const ASI_MAX = 160
   102→   102→function asiAngle(kt: number) {
   103→   103→  return -135 + (Math.min(Math.max(kt, 0), ASI_MAX) / ASI_MAX) * 270
   104→   104→}
   105→   105→// Altimeter 100s needle: full turn per 1000ft, continuous (no modulo) so it
   106→   106→// never snaps backward at the 999->1000 boundary.
   107→   107→function altAngle(ft: number) {
   108→   108→  return (ft / 1000) * 360
   109→   109→}
   110→   110→// VSI: 0 at 9 o'clock (-90deg), +2000 at +50, -2000 at -230
   111→   111→function vsiAngle(fpm: number) {
   112→   112→  const c = Math.max(-2000, Math.min(2000, fpm))
   113→   113→  return -90 + (c / 2000) * 140
   114→   114→}
   115→   115→// DG card rotates so current heading sits under the top lubber pointer.
   116→   116→function dgCardAngle(h: number) {

   132→   132→function polar(r: number, angleDeg: number) {
   133→   133→  const rad = (angleDeg * Math.PI) / 180
   134→   134→  return { x: 50 + r * Math.sin(rad), y: 50 - r * Math.cos(rad) }
   135→   135→}
   136→   136→
   137→   137→function arcPath(r: number, a1: number, a2: number) {
   138→   138→  const p1 = polar(r, a1)
   139→   139→  const p2 = polar(r, a2)
   140→   140→  const large = Math.abs(a2 - a1) > 180 ? 1 : 0
   141→   141→  return `M ${p1.x.toFixed(2)} ${p1.y.toFixed(2)} A ${r} ${r} 0 ${large} 1 ${p2.x.toFixed(2)} ${p2.y.toFixed(2)}`
   142→   142→}

   389→   389→
   390→   390→function Hotspot({
   391→   391→  hotspot,
   392→   392→  selectedId,
   393→   393→  activeCategory,
   394→   394→  onSelect,
   395→   395→  className,
   396→   396→  children,
   397→   397→}: {
   398→   398→  hotspot: Hotspot
   399→   399→  selectedId: string | null
   400→   400→  activeCategory: Category | null
   401→   401→  onSelect: (id: string) => void
   402→   402→  className?: string
   403→   403→  children: ReactNode
   404→   404→}) {
   405→   405→  const dim = activeCategory !== null && activeCategory !== hotspot.category
   406→   406→  const active = selectedId === hotspot.id
   407→   407→  return (
   408→   408→    <div
   409→   409→      role="button"
   410→   410→      tabIndex={0}
   411→   411→      aria-label={`${hotspot.name} — ${hotspot.category}`}
   412→   412→      aria-pressed={active}
   413→   413→      onClick={() => onSelect(hotspot.id)}
   414→   414→      onKeyDown={(e) => {
   415→   415→        if (e.key === 'Enter' || e.key === ' ') {
   416→   416→          e.preventDefault()
   417→   417→          onSelect(hotspot.id)
   418→   418→        }
   419→   419→      }}
   420→   420→      className={`hotspot ${dim ? 'hotspot--dimmed' : ''} ${active ? 'hotspot--active' : ''} ${className ?? ''}`}
