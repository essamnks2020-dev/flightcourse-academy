/* UPGRADE: digit runs from speech recognition ("172", "43", "95") are
   expanded digit-by-digit so "Cessna 172 Bravo" scores as
   "cessna one seven two bravo". */
function tokenize(str){
  let s = ' ' + String(str).toLowerCase() + ' ';
  s = s.replace(/([a-z])(\d)/g,'$1 $2').replace(/(\d)([a-z])/g,'$1 $2');
  s = s.replace(/[^a-z0-9 ]/g,' ');
  const out = [];
  s.split(/\s+/).filter(Boolean).forEach(t => {
    if(/^\d+$/.test(t)) t.split('').forEach(d => out.push(DIGIT[d]));
    else out.push(t);
  });
  return out;
}
function findSeq(hay, needle, start){
  let j = 0, first = -1;
  for(let i = start; i < hay.length && j < needle.length; i++){
    if(hay[i] === needle[j]){ if(j === 0) first = i; j++; }
  }
  return j === needle.length ? first : -1;
}
function align(a, b){
  const n = a.length, m = b.length;
  const dp = Array.from({length:n+1}, () => new Uint16Array(m+1));
  for(let i = n-1; i >= 0; i--) for(let j = m-1; j >= 0; j--)
    dp[i][j] = a[i] === b[j] ? dp[i+1][j+1]+1 : Math.max(dp[i+1][j], dp[i][j+1]);
  const aOk = new Array(n).fill(false), bOk = new Array(m).fill(false);
  let i = 0, j = 0;
  while(i < n && j < m){
    if(a[i] === b[j]){ aOk[i] = bOk[j] = true; i++; j++; }
    else if(dp[i+1][j] >= dp[i][j+1]) i++; else j++;
  }
  return {aOk, bOk, lcs: dp[0][0]};
}
function buildReq(s){
  const ap = s.ap.toLowerCase().split(' ');
  const els = [
    {label:`Open — “${s.ap} traffic”`, tokens:[...ap,'traffic']},

    {label:'Position report', tokens:POS_TOKENS[s.pos](s)},
  ];
  if(!s.noRwy) els.push({label:`Runway — “${s.rwySp}”`, tokens:['runway',...s.rwySp.split(' ')]});
  if(s.intent) els.push({label:`Intentions — “${s.intent}”`, tokens:s.intent.split(' ')});
  els.push({label:`Close — “${s.ap} traffic”`, tokens:[...ap,'traffic']});
  return els;
}
function checkElements(player, els){
  let cur = 0;
  return els.map(el => {
    const at = findSeq(player, el.tokens, cur);
    if(at >= 0){ cur = at + el.tokens.length; return {...el, ok:true}; }
    const any = findSeq(player, el.tokens, 0);
    return {...el, ok:any >= 0, outOfOrder:any >= 0};
  });
}
function scoreCall(raw, s, elapsed){
  const player = tokenize(raw), ref = tokenize(s.ref);
  const notes = [];
  const els = buildReq(s);
  if(!player.length){
    return {overall:0, dims:{phraseology:0,brevity:0,phonetics:0,timing:0,completeness:0},
      notes:['No transmission received. Key the mic and make the call.'], player, ref, al:null,
      els:els.map(e=>({...e,ok:false}))};
  }
  const joined = ' ' + player.join(' ') + ' ';
  const forbidden = joined.includes(' any traffic in the area ');
  const elRes = checkElements(player, els);
  const foundN = elRes.filter(e => e.ok || e.outOfOrder).length;
  const completeness = Math.round(foundN / elRes.length * 100);
  elRes.forEach(e => {
    if(!e.ok && !e.outOfOrder) notes.push('Missing: ' + e.label + '.');
    else if(e.outOfOrder) notes.push('Out of order: ' + e.label + ' — sequence matters on the radio.');
  });
  const al = align(player, ref);
  const phraseology = Math.round(2 * al.lcs / (player.length + ref.length) * 100);
  const extra = Math.max(0, player.length - ref.length);
  let brevity = Math.max(0, 100 - Math.max(0, extra - 2) * 9);
  if(forbidden){
    brevity = Math.min(brevity, 15);
    notes.push('Never say “any traffic in the area, please advise.” AC 90-66C: state who you are, where you are, and your intentions — the CTAF is not a request line.');
  } else if(extra > 4){
    const extras = player.filter((_,i)=>!al.aOk[i]).slice(0,5).join(' ');
    notes.push(`Tighten it — ${extra} extra word${extra>1?'s':''}${extras?` (“${extras}”)`:''}.`);
  }
  let phT = 0, phO = 0;
  (s.phon || []).forEach(p => {
    phT++;
    if(player.includes(p.good)) phO++;
    else notes.push(`Phonetics: ${p.ctx} — say “${p.good}”, not “${p.bad}”.`);
  });
  phT++;
  if(joined.includes(' ' + s.rwyBad + ' ')) notes.push(`Runway numbers go digit by digit — “runway ${s.rwySp}”, never “${s.rwyBad}”.`);
  else phO++;
  const phonetics = Math.round(phO / phT * 100);
  const timing = elapsed <= 5 ? 100 : elapsed >= 28 ? 0 : Math.round(100 * (1 - (elapsed - 5) / 23));
  if(elapsed > 12) notes.push(`Anticipate the call — you keyed up in ${elapsed.toFixed(1)}s. Brief it before you need it.`);
  const dims = {phraseology, brevity, phonetics, timing, completeness};
  let overall = Math.round(phraseology*.3 + completeness*.25 + brevity*.15 + phonetics*.15 + timing*.15);
  if(S.hintUsed){ overall = Math.max(0, overall - 10); notes.push('Structure hint used (−10). Next time brief it from memory.'); }
  if(!notes.length) notes.push('Clean call. An examiner would just nod.');
  return {overall, dims, notes, player, ref, al, els: elRes};
}

  });

  else AU.stinger(res.overall >= 75);

function recip(rwy){ return String((parseInt(rwy,10) + 18) % 36).padStart(2,'0'); }
function drawArrow(ctx,x,y,ang,color){
  ctx.save(); ctx.translate(x,y); ctx.rotate(ang);
  ctx.fillStyle = color;
  ctx.beginPath(); ctx.moveTo(5,0); ctx.lineTo(-3,3.5); ctx.lineTo(-3,-3.5); ctx.closePath(); ctx.fill();
  ctx.restore();
}
function drawPlane(ctx,x,y,hdg,color,glow){
  ctx.save(); ctx.translate(x,y); ctx.rotate(hdg * Math.PI/180);
  if(glow){ ctx.shadowColor = 'oklch(0.79 0.152 74)'; ctx.shadowBlur = 14; }
  ctx.fillStyle = color;
  ctx.beginPath(); ctx.moveTo(0,-7); ctx.lineTo(5,6); ctx.lineTo(0,3); ctx.lineTo(-5,6); ctx.closePath(); ctx.fill();
  ctx.restore();
}
function loopPos(loop, mir, u, t){
  const pts = loop.map(p => [p[0]*u, p[1]*u*mir]);
  const segs = []; let tot = 0;
  for(let i = 0; i < pts.length-1; i++){
    const l = Math.hypot(pts[i+1][0]-pts[i][0], pts[i+1][1]-pts[i][1]);
    segs.push(l); tot += l;
  }
  let d = ((t % 1) + 1) % 1 * tot;
  for(let i = 0; i < segs.length; i++){
    if(d <= segs[i]){
      const f = d / segs[i];
      const x = pts[i][0] + (pts[i+1][0]-pts[i][0]) * f;
      const y = pts[i][1] + (pts[i+1][1]-pts[i][1]) * f;
      const hdg = (Math.atan2(pts[i+1][0]-pts[i][0], -(pts[i+1][1]-pts[i][1])) * 180/Math.PI + 360) % 360;
      return [x, y, hdg];
    }
    d -= segs[i];
  }
  return [pts[0][0], pts[0][1], 270];
}
function drawMap(now){
  const c = $('mapCanvas');
  const r = c.getBoundingClientRect();
  if(r.width < 10) return;
  const d = Math.min(devicePixelRatio || 1, 2);
  if(c.width !== Math.round(r.width*d) || c.height !== Math.round(r.height*d)){
    c.width = Math.round(r.width*d); c.height = Math.round(r.height*d);
  }
  const ctx = c.getContext('2d');
  ctx.setTransform(d,0,0,d,0,0);
  const w = r.width, h = r.height;
  ctx.fillStyle = 'oklch(0.11 0.016 254)'; ctx.fillRect(0,0,w,h);
  ctx.strokeStyle = 'oklch(0.99 0.01 250 / 4%)'; ctx.lineWidth = 1;
  ctx.beginPath();
  for(let x = 0; x < w; x += 28){ ctx.moveTo(x,0); ctx.lineTo(x,h); }
  for(let y = 0; y < h; y += 28){ ctx.moveTo(0,y); ctx.lineTo(w,y); }
  ctx.stroke();
  const s = S.sc; if(!s) return;
  const mir = s.dir === 'right' ? -1 : 1;
  const u = Math.min(w, h) * 0.17;
  ctx.save();
  ctx.translate(w/2, h*0.54);
  ctx.rotate((s.hdg - 270) * Math.PI/180);
  const L = p => [p[0]*u, p[1]*u*mir];
  const loop = [[1,0],[-1.7,0],[-1.7,1.1],[2.3,1.1],[2.3,0],[1,0]];
  ctx.setLineDash([5,5]); ctx.strokeStyle = 'oklch(0.75 0.128 205 / 32%)'; ctx.lineWidth = 1.2;
  ctx.beginPath();
  loop.forEach((p,i)=>{ const q = L(p); i ? ctx.lineTo(q[0],q[1]) : ctx.moveTo(q[0],q[1]); });
  ctx.stroke(); ctx.setLineDash([]);
  for(let i = 0; i < loop.length-1; i++){
    const a = L(loop[i]), b = L(loop[i+1]);
    drawArrow(ctx, (a[0]+b[0])/2, (a[1]+b[1])/2, Math.atan2(b[1]-a[1], b[0]-a[0]), 'oklch(0.75 0.128 205 / 45%)');

  }
  const rw = u * 0.22;
  ctx.fillStyle = 'oklch(0.28 0.02 254)';
  ctx.beginPath(); ctx.roundRect(-u, -rw/2, 2*u, rw, 3); ctx.fill();
  ctx.strokeStyle = 'oklch(0.99 0.01 250 / 25%)'; ctx.lineWidth = 1; ctx.stroke();
  ctx.setLineDash([u*0.12, u*0.09]); ctx.strokeStyle = 'oklch(0.99 0.01 250 / 55%)'; ctx.lineWidth = 1.5;
  ctx.beginPath(); ctx.moveTo(-u*0.9, 0); ctx.lineTo(u*0.9, 0); ctx.stroke(); ctx.setLineDash([]);
  ctx.fillStyle = 'oklch(0.99 0.01 250 / 70%)';
  ctx.font = `700 ${Math.round(u*0.22)}px "JetBrains Mono", monospace`;
  ctx.save(); ctx.translate(u*0.7, 0); ctx.rotate(-Math.PI/2); ctx.fillText(s.rwy, 0, 0); ctx.restore();
  ctx.save(); ctx.translate(-u*0.7, 0); ctx.rotate(Math.PI/2); ctx.fillText(recip(s.rwy), 0, 0); ctx.restore();
  const pos = POS[s.pos];
  const tp = L(pos.p);
  if(s.pos === 'entering-downwind'){
    const from = L([-1.95, 1.9]);
    ctx.setLineDash([4,4]); ctx.strokeStyle = 'oklch(0.79 0.152 74 / 50%)'; ctx.lineWidth = 1.2;
    ctx.beginPath(); ctx.moveTo(from[0], from[1]); ctx.lineTo(tp[0], tp[1]); ctx.stroke(); ctx.setLineDash([]);
    drawArrow(ctx, tp[0], tp[1], Math.atan2(tp[1]-from[1], tp[0]-from[0]), 'oklch(0.79 0.152 74 / 70%)');
  }
  if(s.pos === 'clear-of-runway'){
    const a = L([0.45, 0]);
    ctx.strokeStyle = 'oklch(0.99 0.01 250 / 25%)'; ctx.lineWidth = 3;
    ctx.beginPath(); ctx.moveTo(a[0], a[1]); ctx.lineTo(tp[0], tp[1]); ctx.stroke();
  }
  if(s.pos === 'going-around'){
    const a = L([0.9, 0]), b = L([-1.5, 0]);
    ctx.setLineDash([4,4]); ctx.strokeStyle = 'oklch(0.79 0.152 74 / 45%)'; ctx.lineWidth = 1.2;
    ctx.beginPath(); ctx.moveTo(a[0], a[1]); ctx.lineTo(b[0], b[1]); ctx.stroke(); ctx.setLineDash([]);
  }
  const pu = (Math.sin(now/300) + 1) / 2;
  ctx.strokeStyle = `oklch(0.79 0.152 74 / ${Math.round(35 + pu*35)}%)`;
  ctx.lineWidth = 1.5;
  ctx.beginPath(); ctx.arc(tp[0], tp[1], u*(0.16 + pu*0.07), 0, 7); ctx.stroke();
  TRAFFIC.forEach(tr => {
    const p = loopPos(loop, mir, u, now/1000 * tr.sp + tr.off);
    drawPlane(ctx, p[0], p[1], p[2], 'oklch(0.75 0.128 205 / 65%)', false);
    if(S.mapLabels){
      ctx.fillStyle = 'oklch(0.75 0.128 205 / 50%)';

      ctx.fillText(tr.tag, p[0]+8, p[1]-6);
    }
  });
  /* own ship — ramp → target → hold */
  const hdg = mir === 1 ? pos.h : (360 - pos.h) % 360;

  } else if(S.plane.state === 'move'){
    const f = Math.min(1, (now - S.plane.t0) / 2600);
    const e = 1 - Math.pow(1 - f, 3);
    px = rp[0] + (tp[0] - rp[0]) * e;
    py = rp[1] + (tp[1] - rp[1]) * e;
    let dh = ((hdg - rampHdg + 540) % 360) - 180;
    ph = rampHdg + dh * e;
    if(f >= 1) S.plane.state = 'hold';
  } else {
    px = tp[0]; py = tp[1]; ph = hdg;
  }
  drawPlane(ctx, px, py, ph, 'oklch(0.79 0.152 74)', true);
  ctx.restore();
  /* data block */
  ctx.fillStyle = 'oklch(0.75 0.128 205 / 75%)';
  ctx.font = '9px "JetBrains Mono", monospace'; ctx.textAlign = 'left'; ctx.textBaseline = 'alphabetic';

  ctx.fillStyle = 'oklch(0.79 0.152 74 / 75%)';
  ctx.textAlign = 'right';
  ctx.fillText(s.cs.toUpperCase(), w - 10, 16);
}

/* ---------------- radar canvas ---------------- */
function drawRadar(now){
  const c = $('radarCanvas');
  const r = c.getBoundingClientRect();
  if(r.width < 10) return;
  const d = Math.min(devicePixelRatio || 1, 2);
  if(c.width !== Math.round(r.width*d) || c.height !== Math.round(r.height*d)){
    c.width = Math.round(r.width*d); c.height = Math.round(r.height*d);
  }
  const ctx = c.getContext('2d');
  ctx.setTransform(d,0,0,d,0,0);
  const w = r.width, h = r.height;
  ctx.clearRect(0,0,w,h);
  const res = S.lastRes; if(!res) return;
  const vals = [res.dims.phraseology, res.dims.brevity, res.dims.phonetics, res.dims.timing, res.dims.completeness];
  const labels = ['PHR','BRV','PHO','TIM','CMP'];
  const N = 5, cx = w/2, cy = h/2 + 4, R = Math.min(w,h)/2 - 24;
  const ang = i => -Math.PI/2 + i * 2*Math.PI/N;
  const t = Math.min(1, (now - S.radarT0) / 900);
  const e = 1 - Math.pow(1 - t, 3);
  for(let ring = 1; ring <= 4; ring++){
    ctx.beginPath();
    for(let i = 0; i <= N; i++){
      const a = ang(i % N), rr = R * ring / 4;
      const x = cx + Math.cos(a)*rr, y = cy + Math.sin(a)*rr;
      i ? ctx.lineTo(x,y) : ctx.moveTo(x,y);
    }
    ctx.strokeStyle = 'oklch(0.99 0.01 250 / 7%)'; ctx.lineWidth = 1; ctx.stroke();
  }
  ctx.font = '8px "JetBrains Mono", monospace'; ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
  for(let i = 0; i < N; i++){
    const a = ang(i);
    ctx.beginPath(); ctx.moveTo(cx,cy); ctx.lineTo(cx + Math.cos(a)*R, cy + Math.sin(a)*R);
    ctx.strokeStyle = 'oklch(0.99 0.01 250 / 10%)'; ctx.stroke();
    ctx.fillStyle = 'oklch(0.65 0.02 250 / 70%)';
    ctx.fillText(labels[i], cx + Math.cos(a)*(R+13), cy + Math.sin(a)*(R+13));
  }
  ctx.beginPath();
  for(let i = 0; i <= N; i++){
    const a = ang(i % N), rr = R * (vals[i % N]/100) * e;
    const x = cx + Math.cos(a)*rr, y = cy + Math.sin(a)*rr;
    i ? ctx.lineTo(x,y) : ctx.moveTo(x,y);
  }
  ctx.closePath();
  ctx.fillStyle = 'oklch(0.75 0.128 205 / 18%)'; ctx.fill();
  ctx.strokeStyle = 'oklch(0.75 0.128 205 / 85%)'; ctx.lineWidth = 1.5;
  ctx.shadowColor = 'oklch(0.75 0.128 205)'; ctx.shadowBlur = 8;
  ctx.stroke(); ctx.shadowBlur = 0;
  for(let i = 0; i < N; i++){
    const a = ang(i), rr = R * (vals[i]/100) * e;
    ctx.beginPath();
    ctx.arc(cx + Math.cos(a)*rr, cy + Math.sin(a)*rr, 2.6, 0, 7);
    ctx.fillStyle = 'oklch(0.79 0.152 74)';
    ctx.shadowColor = 'oklch(0.79 0.152 74)'; ctx.shadowBlur = 6;
    ctx.fill(); ctx.shadowBlur = 0;
  }
}

/* ---------------- main render loop ---------------- */
function tick(now){
  if(S.screen === 'brief' || S.screen === 'live' || S.screen === 'feed') drawMap(now);
  if(S.screen === 'feed'){
    drawRadar(now);
    if(S.scoreTarget != null){
      const t = Math.min(1, (now - S.scoreT0) / 900);
      const e = 1 - Math.pow(1 - t, 3);
      $('scoreBig').textContent = Math.round(S.scoreTarget * e);
    }
  }
  requestAnimationFrame(tick);
}
